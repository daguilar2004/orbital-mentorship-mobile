const loadMenteeNamesForMentorshipsOnce = useCallback(async (mentorshipIds) => {
  const ids = [...new Set((mentorshipIds || []).filter(Boolean))];
  if (!ids.length) return [];

  // one token for the whole batch
  const headers = await getAuthHeaders();

  // small concurrency to avoid connection exhaustion
  const fetchJson = async (url) => {
    const res = await fetch(url, { method: 'GET', headers });
    if (!res.ok) throw new Error(`${res.status} ${res.statusText}`);
    return res.json();
  };

  // 1) mentorships → menteeIds (limit parallelism)
  const msResults = [];
  const MS_CHUNK = 5;
  for (let i = 0; i < ids.length; i += MS_CHUNK) {
    const chunk = ids.slice(i, i + MS_CHUNK);
    const part = await Promise.all(
      chunk.map(id => fetchJson(`${ORB_SERVER_API_URL}/api/mentorships/${id}`).catch(() => null))
    );
    msResults.push(...part);
  }
  const pairs = ids.map((id, i) => ({ mentorshipId: id, menteeId: msResults[i]?.data?.menteeId ?? msResults[i]?.menteeId ?? null }))
                  .filter(p => p.menteeId);

  if (!pairs.length) return [];

  // 2) mentees → names (limit parallelism)
  const namesMap = new Map();
  const M_CHUNK = 5;
  for (let i = 0; i < pairs.length; i += M_CHUNK) {
    const chunk = pairs.slice(i, i + M_CHUNK);
    const part = await Promise.all(
      chunk.map(p => fetchJson(`${ORB_SERVER_API_URL}/api/mentees/${p.menteeId}`).catch(() => null))
    );
    part.forEach((mentee, idx) => {
      const mId = chunk[idx].menteeId;
      if (mentee) namesMap.set(mId, mentee.name ?? 'Unknown');
    });
  }

  return pairs.map(p => ({
    mentorshipId: p.mentorshipId,
    menteeId: p.menteeId,
    name: namesMap.get(p.menteeId) ?? 'Unknown'
  }));
}, [getAuthHeaders, ORB_SERVER_API_URL]);

    const id = user?.id || user?._id;

    function computeDisplayFromUser(user) {
        if (!user) return { name: 'Unknown', avatarUrl: '' };
        const name = [user.firstName, user.lastName].filter(Boolean).join(' ') || user.username || 'Unknown';
        const avatarUrl = user.picture || user.profilePictureUrl || '';
        return { name, avatarUrl };
    }

    function pushDisplayForUser(userId, userObj){
        const display = computeDisplayFromUser(userObj);
        setDisplayByUserId(s => ({...s,[userId]: {...display, lastComputed: Date.now() }}))
    }

    async function getDisplayByUserId(userId, { force = false } = {}) {
        // if we already computed and user cache is fresh, return it
        const disp = displayByUserId[userId];
        const userEntry = usersByUserId[userId];

        const freshUser = userEntry && shouldUseCache(userEntry);
        const freshDisp  = disp && freshUser && !force; // if user is fresh, display is implicitly fresh

        if (freshDisp) return disp;

        // ensure user is cached
        const user = await fetchUserIfNeeded(userId, { force });
        const nextDisplay = computeDisplayFromUser(user);
        setDisplayByUserId(s => ({ ...s, [userId]: { ...nextDisplay, lastComputed: Date.now() } }));
        return nextDisplay;

    }

     * Fetches user identity information from the server

     *

     * - Retrieves data using Auth0 token for authentication
     * - Handles 404 gracefully for new users

     * 
     * @returns {Promise<Object|null>} The user data object or null if not available/error

    const fetchUserIds = useCallback(async () => {
        try {
            if (!isAuthenticated || !user) {
                setIsContextLoading(false)
                return null
            }
            
            setIsContextLoading(true);
            
            const auth0Token = await getAccessTokenSilently({
                authorizationParams: {
                    audience: import.meta.env.VITE_APP_AUTH0_AUDIENCE,
                },
            });

            const response = await fetch(`${ORB_SERVER_API_URL}/api/users/my-user`, {
                method: 'GET',
                headers: {
                    Authorization: `Bearer ${auth0Token}`,
                    'Content-Type': 'application/json',
                },
            });

            if (!response.ok) {
                if (response.status === 404) {
                    // User document not found, which is normal for new users
                    console.log('User document not found - this is normal for new users')
                    setIsContextLoading(false)
                    return null
                }
                throw new Error(`HTTP error! Status: ${response.status}`)
            }

            const data = await response.json();
            setUserId(data._id || '');
            setRole(data.role || '');
           
            
            // Ensure proper parsing of the IDs, which might be MongoDB ObjectIDs
            const parsedMenteeId = (
                data.menteeId && typeof data.menteeId.toString === 'function'
                )
                ? data.menteeId.toString()
                : '';
            const parsedMentorId = data.mentorId ? data.mentorId.toString() : '';
            
            console.log('UserContext: Fetched user data:', {
                _id: data._id,
                role: data.role,
                menteeId: parsedMenteeId,
                mentorId: parsedMentorId,
                setupProgress: data.setupProgress
            });
            
            setMenteeId(parsedMenteeId);
            setMentorId(parsedMentorId);
            setIsContextLoading(false);
            setProgress(data.setupProgress || 1); // adds the setUpProgrss number to the user context
            
            return data  // Return the data so caller can use it directly
        }
        catch (error) {
            console.error('Error fetching user data:', error)
            setUserId('')
            setRole('')
            setMentorId('')
            setMenteeId('')
            setIsContextLoading(false)
            return null
        }
    }, [isAuthenticated, user, getAccessTokenSilently, ORB_SERVER_API_URL])

    /**
     * Fetches active mentorship IDs based on the user's role
     * For mentors: retrieves all current mentorships (potentially multiple)
     * For mentees: retrieves the current mentorship (single)
     * 
     * @param {string} currentRole - The user's role ('mentor' or 'mentee')
     * @param {string} currentMentorId - The user's mentor ID (if they are a mentor)
     * @param {string} currentMenteeId - The user's mentee ID (if they are a mentee)
     * @returns {Promise<void>}
     */
    const fetchMentorshipIds = useCallback(async (currentRole, currentMentorId, currentMenteeId) => {
        try {
            if (!isAuthenticated || !user || !currentRole) {
                return
            }
            
            // Guard against invalid IDs
            if (currentRole === 'mentor' && !currentMentorId) {
                console.log('Cannot fetch mentorships: No mentor ID available')
                return
            }
            
            if (currentRole === 'mentee' && !currentMenteeId) {
                console.log('Cannot fetch mentorships: No mentee ID available')
                return
            }
            
            const auth0Token = await getAccessTokenSilently({
                authorizationParams: {
                    audience: import.meta.env.VITE_APP_AUTH0_AUDIENCE,
                },
            })

            const idToUse = currentRole === 'mentee' ? currentMenteeId : currentMentorId
            
            const response = await fetch(`${ORB_SERVER_API_URL}/api/${currentRole}s/${idToUse}`, {
                method: 'GET',
                headers: {
                    'Authorization': `Bearer ${auth0Token}`,
                    'Content-Type': 'application/json',
                },
            })

            if (!response.ok) {
                throw new Error(`HTTP eror! Status: ${response.status}`)
            }

            const data = await response.json()

            if (currentRole === "mentor") {
                setMentorshipIds(data.currentMentorships || [])
            }
            else if (currentRole === "mentee") {
                setMentorshipIds(data.currentMentorship ? [data.currentMentorship] : [])
            }
        }
        catch (error) {
            console.error('Error fetching mentorship data:', error)
            setMentorshipIds([''])
        }
    }, [isAuthenticated, user, getAccessTokenSilently, ORB_SERVER_API_URL])

    /**
     * Resets and refreshes all user context data
     * Used when user data may have changed and needs to be re-fetched
     * 
     * The process:
     * 1. Clear all state variables
     * 2. Fetch basic user information
     * 3. Fetch mentorship information based on updated user data
     * 
     * @returns {Promise<boolean>} Success status of the refresh operation
     */
    const refreshUserContext = useCallback(async () => {
        try {
            console.log("Refreshing user context...")
            
            setIsContextLoading(true)
            setUserId('')
            setRole('')
            setMentorId('')
            setMenteeId('')
            setMentorshipIds([''])
            
            // Fetch basic user info
            await fetchUserIds()
            
            // Short delay to ensure state updates have propagated
            await new Promise(resolve => setTimeout(resolve, 200))
            
            // Get the latest values after state updates
            const latestRole = role
            const latestMentorId = mentorId
            const latestMenteeId = menteeId
            
            if (latestRole && (latestMentorId || latestMenteeId)) {
                await fetchMentorshipIds(latestRole, latestMentorId, latestMenteeId)
            }
            
            console.log("User context refreshed successfully")
            setIsContextLoading(false)
            return true
        }
        catch (error) {
            console.error('Error refreshing user context:', error)
            setIsContextLoading(false)
            return false
        }
    }, [fetchUserIds, fetchMentorshipIds, role, mentorId, menteeId])

    }, [isLoading, isAuthenticated, menteeId, mentorId])

    /**
     * Effect to initialize user context when authentication state changes
     * Fetches user IDs on app startup or when auth state changes
     * Clears user data when not authenticated
     */
    useEffect(() => {
        if (!isLoading && isAuthenticated) {
            fetchUserIds()
        } else if (!isAuthenticated) {
            // Clear user data when not authenticated
            setUserId('')
            setRole('')
            setMentorId('')
            setMenteeId('')
            setMentorshipIds([''])
            setIsContextLoading(false)
        }
    }, [isLoading, isAuthenticated, fetchUserIds])

    /**
     * Effect to fetch mentorship IDs once user identity and role are established
     * Only executes when we have the required IDs based on the user's role
     */
    useEffect(() => {
        const hasRequiredId = (role === 'mentor' && mentorId) || (role === 'mentee' && menteeId)
        
        if (!isLoading && isAuthenticated && role && hasRequiredId) {
            fetchMentorshipIds(role, mentorId, menteeId)
        }
    }, [isLoading, isAuthenticated, role, mentorId, menteeId, fetchMentorshipIds])

    /**
     * Clears all user context state
     * Used during logout or when needing to reset user state
     */
    const clearUserContext = useCallback(() => {
        setUserId('')
        setRole('')
        setMentorId('')
        setMenteeId('')
        setMentorshipIds([''])
    }, [])

   const value = useMemo(() => ({
    // existing ids & flags
    userId, role, mentorId, menteeId, mentorshipIds, isContextLoading, setupProgress,
    clearUserContext, refreshUserContext,
    selectMyMentee,selectMyMentor,
    // caches + helpers
    usersByUserId, menteesById, mentorshipsById,
    fetchUserIfNeeded, fetchMenteeIfNeeded, fetchMentorshipIfNeeded,
    fetchMyMenteeIfNeeded, fetchMyMentorIfNeeded,
    getMenteeNameByMentorshipId,

    fetchMenteeNameList,

    loadMenteeNamesForMentorshipsOnce,

    
    selectUser, selectMentee, selectMentorship, 
    displayNameFromUser, selectDisplayByUserId,selectMenteeNameByMentorshipId,
    getDisplayByUserId,
    pushUserBasic, pushDisplayForUser

    }), [
    userId, role, mentorId, menteeId, mentorshipIds, isContextLoading, setupProgress,
    usersByUserId, menteesById, mentorshipsById,
    fetchUserIfNeeded, fetchMenteeIfNeeded, fetchMentorshipIfNeeded,fetchMyMentorIfNeeded,fetchMenteeNameList
    ]);

    return (
        <UserContext.Provider value={value}>
            {children}
        </UserContext.Provider>
    )
}

/**
 * Custom hook to use the UserContext
 * Validates that the hook is used within a UserProvider
 * 
 * @returns {Object} The user context object containing all user state and methods
 * @throws {Error} If used outside of a UserProvider
 */
export const useUser = () => {
    const context = useContext(UserContext)
    if (context === undefined) {
        throw new Error('useUser must be used within a UserProvider')
    }
    return context
}
