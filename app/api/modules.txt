import axios from 'axios';
import { API_BASE_URL } from "./index";

export const fetchModules = async (role) => {
    try{
        const response = await axois.get(`${API_BASE_URL}/api/modules?targetAudience=${role}`);
        return response.data;

    }catch(error){
        console.error("Error fetching modules:", error);
        throw error;

    }
};
